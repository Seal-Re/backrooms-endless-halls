**全局契约（所有 data 与 engine 共享）**：每个 JS 文件形如
```js
(function (g) {
  const GAME = (g.GAME = g.GAME || {});
  // ... 挂载到 GAME
})(typeof window !== 'undefined' ? window : globalThis);
```
Node 测试用 `createRequire` `require()` 这些文件（副作用挂 `globalThis.GAME`）。**禁用 import/export**，保证 file:// 与 Node 双跑。

**节点契约（content 硬约束，lint 据此校验）**：
- `level = { id, name, class:0..5, danger, safe:bool, entities:[id], nodes:{ [id]: node } }`
- `node = { id, text, variants?:{calm|unease|dread|broken:str|fn}, onEnter?:effects|fn, choices:[choice] }`，**每节点 ≥1 个 choice**。
- `choice = { label, requires?:(s)=>bool, effects?:[[op,...]], goto?:"node"|"level:node", ending?:"endingId" }`，**每 choice 必须有 `goto` 或 `ending`（二选一）**；导航只走 `goto`/`ending`，**不得藏在 effects 里**（`drop` 随机跌层除外）。
- `effects` 原语：`["sanity",±n] ["thirst",±n] ["stamina",±n] ["give",item,n?] ["take",item,n?] ["flag",name,val?] ["trust",±n] ["drop"]`。
- 物品 id 必须存在于 `GAME.items`；ending id 必须存在于 `GAME.endings`。

```js
  GAME.newState = function (startId, seed) {
    const start = GAME.starts[startId];
    if (!start) throw new Error('unknown start: ' + startId);
    return {
      level: start.level, node: start.node,
      sanity: 100, thirst: 0, stamina: 100, turns: 0,
      inv: Object.assign({}, start.inv || {}),
      flags: {}, trust: 0, visited: {}, seed: seed >>> 0,
      redCooldown: 0, ended: null,
    };
  };

  GAME.goto = function (state, target) {
    let level = state.level, node = target;
    const i = target.indexOf(':');
    if (i >= 0) { level = target.slice(0, i); node = target.slice(i + 1); }
    state.level = level; state.node = node;
    const key = level + ':' + node;
    state.visited[key] = (state.visited[key] || 0) + 1;
  };

  GAME.dropLevel = function (state) {
    const rng = GAME.makeRng((state.seed + state.turns) >>> 0);
    const pool = GAME.DROP_POOL && GAME.DROP_POOL.length ? GAME.DROP_POOL : ['L0:entry'];
    GAME.goto(state, pool[Math.floor(rng() * pool.length)]);
  };

  GAME.applyEffects = function (state, effects) {
    if (!effects) return state;
    for (const e of effects) {
      const op = e[0], a = e[1], b = e[2];
      switch (op) {
        case 'sanity': state.sanity = GAME.clamp(state.sanity + a, 0, 100); break;
        case 'thirst': state.thirst = GAME.clamp(state.thirst + a, 0, 100); break;
        case 'stamina': state.stamina = GAME.clamp(state.stamina + a, 0, 100); break;
        case 'give': state.inv[a] = (state.inv[a] || 0) + (b == null ? 1 : b); break;
        case 'take': state.inv[a] = Math.max(0, (state.inv[a] || 0) - (b == null ? 1 : b)); if (!state.inv[a]) delete state.inv[a]; break;
        case 'flag': state.flags[a] = (b == null ? true : b); break;
        case 'trust': state.trust += a; break;
        case 'drop': GAME.dropLevel(state); break;
        default: throw new Error('unknown effect op: ' + op);
      }
    }
    return state;
  };
```

```js
  GAME.tick = function (state) {
    state.turns += 1;
    const rng = GAME.makeRng((state.seed * 7919 + state.turns) >>> 0);
    let drain = 1 + Math.floor(rng() * 3);
    if (state.flags.addicted) drain += 1;
    const lvl = GAME.levels[state.level];
    if (lvl && lvl.safe) drain = -5;
    state.sanity = GAME.clamp(state.sanity - drain, 0, 100);
    state.thirst = GAME.clamp(state.thirst + 2, 0, 100);
    if (state.thirst >= 90) state.stamina = GAME.clamp(state.stamina - 5, 0, 100);
    if (state.sanity <= 0 && !state.ended) state.ended = 'sanity_zero';
    if (state.stamina <= 0 && !state.ended) state.ended = 'collapse';
    return state;
  };

  GAME.currentNode = function (state) {
    const lvl = GAME.levels[state.level];
    if (!lvl) throw new Error('no level ' + state.level);
    const node = lvl.nodes[state.node];
    if (!node) throw new Error('no node ' + state.level + ':' + state.node);
    return node;
  };

  GAME.visibleChoices = function (state) {
    const node = GAME.currentNode(state);
    return (node.choices || []).filter(c => !c.requires || c.requires(state));
  };

  GAME.choose = function (state, idx) {
    const c = GAME.visibleChoices(state)[idx];
    if (!c) throw new Error('bad choice idx ' + idx);
    if (Array.isArray(c.effects)) GAME.applyEffects(state, c.effects);
    if (c.ending) { state.ended = c.ending; return state; }
    if (c.goto) GAME.goto(state, c.goto);
    if (!c.noTick && !state.ended) GAME.tick(state);
    if (!state.ended) {
      const n = GAME.currentNode(state);
      if (n.onEnter) {
        const eff = (typeof n.onEnter === 'function') ? n.onEnter(state) : n.onEnter;
        if (Array.isArray(eff)) GAME.applyEffects(state, eff);
      }
    }
    return state;
  };
```

```js
  GAME.nodeText = function (state) {
    const node = GAME.currentNode(state);
    let t = node.text;
    const tier = GAME.sanityTier(state.sanity);
    if (node.variants && node.variants[tier] != null) t = node.variants[tier];
    if (typeof t === 'function') t = t(state);
    return String(t);
  };

  function b64enc(s) {
    if (typeof btoa !== 'undefined') return btoa(unescape(encodeURIComponent(s)));
    return Buffer.from(s, 'utf8').toString('base64');
  }
  function b64dec(s) {
    if (typeof atob !== 'undefined') return decodeURIComponent(escape(atob(s)));
    return Buffer.from(s, 'base64').toString('utf8');
  }
  GAME.serialize = (state) => b64enc(JSON.stringify(state));
  GAME.deserialize = (str) => JSON.parse(b64dec(str));
```

```js
import { createRequire } from 'node:module';
import { fileURLToPath } from 'node:url';
import { dirname, join } from 'node:path';
import { readdirSync } from 'node:fs';
const require = createRequire(import.meta.url);
const here = dirname(fileURLToPath(import.meta.url));
const load = (p) => require(join(here, p));

load('engine.js');
['data/items.js','data/entities.js','data/orgs.js','data/endings.js','data/starts.js'].forEach(f=>{ try{load(f)}catch(e){} });
const lvlDir = join(here, 'data/levels');
for (const f of readdirSync(lvlDir).filter(f=>f.endsWith('.js'))) load(join('data/levels', f));

const GAME = globalThis.GAME;
const errors = [], warns = [];
const nodeExists = (target, curLevel) => {
  let lv = curLevel, nd = target; const i = target.indexOf(':');
  if (i>=0){ lv = target.slice(0,i); nd = target.slice(i+1); }
  return !!(GAME.levels[lv] && GAME.levels[lv].nodes[nd]);
};

// 1) 每节点 ≥1 choice；每 choice 有 goto 或 ending；goto/ending/item 目标存在
for (const [lid, lv] of Object.entries(GAME.levels)) {
  for (const [nid, node] of Object.entries(lv.nodes)) {
    const where = `${lid}:${nid}`;
    if (!node.choices || !node.choices.length) errors.push(`死节点（无 choice）: ${where}`);
    for (const c of (node.choices||[])) {
      if (!c.goto && !c.ending) errors.push(`choice 缺 goto/ending: ${where} → "${c.label}"`);
      if (c.goto && !nodeExists(c.goto, lid)) errors.push(`悬空 goto: ${where} → ${c.goto}`);
      if (c.ending && !GAME.endings[c.ending]) errors.push(`未知 ending: ${where} → ${c.ending}`);
      for (const e of (c.effects||[])) {
        if ((e[0]==='give'||e[0]==='take') && !GAME.items[e[1]]) errors.push(`未知物品: ${where} → ${e[1]}`);
      }
    }
    for (const e of (node.onEnter && Array.isArray(node.onEnter) ? node.onEnter : [])) {
      if ((e[0]==='give'||e[0]==='take') && !GAME.items[e[1]]) errors.push(`onEnter 未知物品: ${where} → ${e[1]}`);
    }
  }
}
// 2) 引擎内置 ending 必须存在
for (const need of ['sanity_zero','collapse']) if (!GAME.endings[need]) errors.push(`缺内置 ending: ${need}`);
// 3) 起点目标存在
for (const [sid, st] of Object.entries(GAME.starts)) if (!nodeExists(st.node, st.level)) errors.push(`起点目标不存在: ${sid} → ${st.level}:${st.node}`);
// 4) 可达性（BFS，忽略 requires）+ ending 引用统计
const reached = new Set(), reachedEndings = new Set(['sanity_zero','collapse']);
const q = Object.values(GAME.starts).map(s=>`${s.level}:${s.node}`);
while (q.length) {
  const cur = q.shift(); if (reached.has(cur)) continue; reached.add(cur);
  const i = cur.indexOf(':'), lv = cur.slice(0,i), nd = cur.slice(i+1);
  const node = GAME.levels[lv] && GAME.levels[lv].nodes[nd]; if (!node) continue;
  for (const c of (node.choices||[])) {
    if (c.ending) reachedEndings.add(c.ending);
    if (c.goto) { const t = c.goto.indexOf(':')>=0 ? c.goto : `${lv}:${c.goto}`; if (!reached.has(t)) q.push(t); }
  }
}
for (const [lid, lv] of Object.entries(GAME.levels)) for (const nid of Object.keys(lv.nodes))
  if (!reached.has(`${lid}:${nid}`)) warns.push(`不可达节点: ${lid}:${nid}`);
for (const eid of Object.keys(GAME.endings)) if (!reachedEndings.has(eid)) warns.push(`不可达 ending: ${eid}`);

console.log(`lint: ${Object.keys(GAME.levels).length} 层, 错误 ${errors.length}, 警告 ${warns.length}`);
warns.forEach(w=>console.log('  ⚠ '+w));
errors.forEach(e=>console.log('  ✖ '+e));
process.exit(errors.length ? 1 : 0);
```

```js
  const rng = GAME.makeRng(1000 + i);
  let s;
  try { s = GAME.newState(startId, 1000 + i); } catch(e){ crashes++; console.log('newState 崩溃 '+startId+': '+e.message); continue; }
  let steps = 0;
  while (!s.ended && steps < CAP) {
    let choices;
    try { choices = GAME.visibleChoices(s); } catch(e){ crashes++; console.log(`visibleChoices 崩溃 @${s.level}:${s.node}: ${e.message}`); break; }
    if (!choices.length) { deadends++; console.log(`死胡同 @${s.level}:${s.node}（start=${startId} seed=${1000+i}）`); break; }
    try { GAME.choose(s, Math.floor(rng() * choices.length)); } catch(e){ crashes++; console.log(`choose 崩溃 @${s.level}:${s.node}: ${e.message}`); break; }
    steps++;
  }
  if (s.ended) endHist[s.ended] = (endHist[s.ended]||0)+1;
  else if (steps >= CAP) capped++;
}
console.log(`\n漫游 ${WALKS} 局：崩溃 ${crashes}，死胡同 ${deadends}，超步未结束 ${capped}`);
console.log('结局分布：', endHist);
process.exit((crashes || deadends) ? 1 : 0);
```
